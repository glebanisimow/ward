# Ward (standalone + VST3): layers of medical-device sources (../engines/devices/)
# with granular and FX chains. Faust code is generated at build time.
cmake_minimum_required(VERSION 3.22)
project(Ward VERSION 0.3.1 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Static CRT: the exe runs without the VC++ redistributable.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
# Release builds keep PDBs next to the binaries so a crash backtrace names
# functions and lines.
if(MSVC)
    add_compile_options("$<$<CONFIG:Release>:/Zi;/FS>")
    add_link_options("$<$<CONFIG:Release>:/DEBUG;/OPT:REF;/OPT:ICF>")
endif()

set(TOOLS "$ENV{LOCALAPPDATA}/drone-tools")
set(JUCE_DIR "${TOOLS}/JUCE" CACHE PATH "JUCE source tree")
find_program(FAUST faust HINTS "${TOOLS}/faust/bin" REQUIRED)
get_filename_component(FAUST_BIN "${FAUST}" DIRECTORY)
get_filename_component(FAUST_INCLUDE "${FAUST_BIN}/../include" ABSOLUTE)

add_subdirectory(${JUCE_DIR} JUCE)

get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(GEN "${CMAKE_CURRENT_BINARY_DIR}/gen")
file(MAKE_DIRECTORY ${GEN})

set(GEN_HEADERS)
# Ward devices: one Faust class and one generated translation unit each.
set(WARD_DEVICES monitor alarm ventilator pump concentrator doppler defib mri room)
set(DEVICE_SOURCES)
foreach(dev ${WARD_DEVICES})
    add_custom_command(
        OUTPUT ${GEN}/dev_${dev}.h
        COMMAND ${FAUST} -I ${ROOT}/lib -ftz 2 -lang cpp -cn Dev_${dev}
                ${ROOT}/engines/devices/${dev}.dsp -o ${GEN}/dev_${dev}.h
        DEPENDS ${ROOT}/engines/devices/${dev}.dsp ${ROOT}/lib/devices.lib ${ROOT}/lib/contract.lib
        COMMENT "faust devices/${dev}.dsp -> Dev_${dev}")
    list(APPEND GEN_HEADERS ${GEN}/dev_${dev}.h)
    file(CONFIGURE OUTPUT ${GEN}/dev_${dev}.cpp CONTENT
"// Generated by CMake: Faust class Dev_${dev} (engines/devices/${dev}.dsp), own TU.
#include \"FaustBridge.h\"
#include \"dev_${dev}.h\"

std::unique_ptr<dsp> createDev_${dev}() { return std::make_unique<Dev_${dev}>(); }
")
    list(APPEND DEVICE_SOURCES ${GEN}/dev_${dev}.cpp)
endforeach()

# One target owns the generated headers so the Visual Studio generator does not
# run the same custom command twice.
add_custom_target(faust_gen DEPENDS ${GEN_HEADERS})

# Generated Faust code: IEEE floats (the contract polynomials and wraps assume
# them).
set_source_files_properties(${DEVICE_SOURCES} PROPERTIES COMPILE_OPTIONS "/W0;/O2")

function(drone_app target)
    add_dependencies(${target} faust_gen)
    target_include_directories(${target} PRIVATE src ${GEN} ${FAUST_INCLUDE})
    target_compile_definitions(${target} PUBLIC
        JUCE_WEB_BROWSER=0
        JUCE_USE_CURL=0
        JUCE_VST3_CAN_REPLACE_VST2=0
        JUCE_APPLICATION_NAME_STRING="$<TARGET_PROPERTY:${target},JUCE_PRODUCT_NAME>"
        JUCE_APPLICATION_VERSION_STRING="$<TARGET_PROPERTY:${target},JUCE_VERSION>")
    target_link_libraries(${target}
        PRIVATE
            juce::juce_audio_utils
        PUBLIC
            juce::juce_recommended_config_flags
            juce::juce_recommended_lto_flags
            juce::juce_recommended_warning_flags)
endfunction()

juce_add_plugin(Ward
    PRODUCT_NAME "Ward"
    VERSION 0.3.1
    COMPANY_NAME "drone-lattice"
    BUNDLE_ID com.dronelattice.ward
    PLUGIN_MANUFACTURER_CODE Drnl
    PLUGIN_CODE Wrd1
    FORMATS Standalone VST3
    IS_SYNTH TRUE
    NEEDS_MIDI_INPUT TRUE
    NEEDS_MIDI_OUTPUT FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS FALSE
    COPY_PLUGIN_AFTER_BUILD FALSE
    VST3_CATEGORIES Instrument Synth
    VST3_AUTO_MANIFEST FALSE
    ICON_BIG ${CMAKE_CURRENT_SOURCE_DIR}/assets/ward_icon.png)

set(WARD_SOURCES
    src/ward/Devices.cpp
    src/ward/Granular.cpp
    src/ward/Layer.cpp
    src/ward/WardProcessor.cpp
    src/ward/WardEditor.cpp
    src/Recorder.cpp
    ${DEVICE_SOURCES})

target_sources(Ward PRIVATE ${WARD_SOURCES} src/Widgets.cpp src/entry/WardEntry.cpp)
target_include_directories(Ward PRIVATE src/ward)
target_link_libraries(Ward PRIVATE juce::juce_dsp)
drone_app(Ward)
